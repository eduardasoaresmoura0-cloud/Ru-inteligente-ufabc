#include <WiFi.h>
#include <HTTPClient.h>
#include <NTPClient.h>
#include <WiFiUdp.h>
#include <time.h>
#include <math.h>

/*
  RU INTELIGENTE — UFABC
  ESP32 + 2 sensores + Firebase

  Esta versão:
  - conta entradas e saídas;
  - mantém lotação, total de entradas, pico e primeira entrada;
  - registra a lotação em cada entrada/saída;
  - calcula tempo estimado de espera pela média recente entre saídas;
  - sincroniza os contadores com o Firebase ao reiniciar;
  - zera os dados diários quando muda a data.
*/

// ==================== WI-FI ====================
const char* ssid = "Galaxy A17 B973";
const char* password = "lucas123";

// ==================== FIREBASE ====================
const String BASE =
  "https://smart-ru-4dbc4-default-rtdb.firebaseio.com/RU";

const String urlLotacao         = BASE + "/lotacao.json";
const String urlTotalEntrada    = BASE + "/totalEntrada.json";
const String urlPicoDia         = BASE + "/picoDia.json";
const String urlPrimeiraEntrada = BASE + "/primeiraEntrada.json";
const String urlTempoEspera     = BASE + "/tempoEspera.json";
const String urlDataAtual       = BASE + "/dataAtual.json";

const String urlHistoricoHora   = BASE + "/historicoPorHora/";
const String urlHistoricoEvento = BASE + "/historicoLotacao/";

// ==================== NTP ====================
// Brasília = UTC-3
WiFiUDP ntpUDP;
NTPClient timeClient(ntpUDP, "pool.ntp.org", -10800, 60000);

// ==================== PINOS ====================
const int SENSOR_ENTRADA = 22;
const int SENSOR_SAIDA   = 23;

// ==================== CONFIGURAÇÕES ====================
const int CAPACIDADE = 50;

// Guardamos até 6 saídas para calcular até 5 intervalos.
const int MAX_SAIDAS = 6;

// Intervalos maiores que 30 minutos não entram na média.
const unsigned long LIMITE_INTERVALO_SAIDA = 30UL * 60UL;

// ==================== VARIÁVEIS ====================
int lotacaoRU = 0;
int totalEntrada = 0;
int picoDia = 0;

int estadoCatraca = 0;
unsigned long inicioEstadoMs = 0;

bool primeiraEntradaRegistrada = false;

unsigned long saidas[MAX_SAIDAS];
int qtdSaidas = 0;

int ultimaEsperaEnviada = -999;
String dataLocal = "";

// =====================================================
// UTILITÁRIOS
// =====================================================

String doisDigitos(int n) {
  return n < 10 ? "0" + String(n) : String(n);
}

bool wifiOK() {
  return WiFi.status() == WL_CONNECTED;
}

void garantirWiFi() {
  if (wifiOK()) return;

  WiFi.disconnect();
  WiFi.begin(ssid, password);

  unsigned long inicio = millis();

  while (!wifiOK() && millis() - inicio < 15000) {
    delay(300);
  }
}

// =====================================================
// FIREBASE
// =====================================================

bool firebasePutInt(const String& url, int valor) {
  garantirWiFi();
  if (!wifiOK()) return false;

  HTTPClient http;
  http.begin(url);
  http.addHeader("Content-Type", "application/json");

  int codigo = http.PUT(String(valor));
  http.end();

  return codigo >= 200 && codigo < 300;
}

bool firebasePutString(const String& url, const String& valor) {
  garantirWiFi();
  if (!wifiOK()) return false;

  HTTPClient http;
  http.begin(url);
  http.addHeader("Content-Type", "application/json");

  String json = "\"" + valor + "\"";
  int codigo = http.PUT(json);

  http.end();

  return codigo >= 200 && codigo < 300;
}

String firebaseGet(const String& url) {
  garantirWiFi();
  if (!wifiOK()) return "";

  HTTPClient http;
  http.begin(url);

  int codigo = http.GET();

  if (codigo < 200 || codigo >= 300) {
    http.end();
    return "";
  }

  String resposta = http.getString();
  http.end();

  resposta.trim();
  return resposta;
}

int firebaseGetInt(const String& url, int padrao = 0) {
  String valor = firebaseGet(url);

  if (valor == "" || valor == "null") {
    return padrao;
  }

  return valor.toInt();
}

String firebaseGetString(const String& url) {
  String valor = firebaseGet(url);

  if (valor == "" || valor == "null") {
    return "";
  }

  if (valor.startsWith("\"") && valor.endsWith("\"")) {
    valor = valor.substring(1, valor.length() - 1);
  }

  return valor;
}

bool firebaseDelete(const String& url) {
  garantirWiFi();
  if (!wifiOK()) return false;

  HTTPClient http;
  http.begin(url);

  int codigo = http.sendRequest("DELETE");
  http.end();

  return codigo >= 200 && codigo < 300;
}

// =====================================================
// DATA / HORA
// =====================================================

bool atualizarHora() {
  if (!wifiOK()) return false;

  if (timeClient.update()) {
    return true;
  }

  return timeClient.forceUpdate();
}

String obterData() {
  time_t epoch = timeClient.getEpochTime();
  struct tm* t = gmtime(&epoch);

  if (!t) return "";

  char buffer[11];

  snprintf(
    buffer,
    sizeof(buffer),
    "%04d-%02d-%02d",
    t->tm_year + 1900,
    t->tm_mon + 1,
    t->tm_mday
  );

  return String(buffer);
}

String obterHoraMinuto() {
  return
    doisDigitos(timeClient.getHours()) +
    ":" +
    doisDigitos(timeClient.getMinutes());
}

String chaveEvento() {
  // Exemplo: 183742 = 18:37:42
  return
    doisDigitos(timeClient.getHours()) +
    doisDigitos(timeClient.getMinutes()) +
    doisDigitos(timeClient.getSeconds());
}

// =====================================================
// HISTÓRICO
// =====================================================

void registrarHistorico() {
  if (!atualizarHora()) return;

  // Histórico detalhado usado pelo novo gráfico.
  String urlEvento =
    urlHistoricoEvento +
    chaveEvento() +
    ".json";

  firebasePutInt(urlEvento, lotacaoRU);

  // Mantém o histórico antigo por hora para compatibilidade.
  String urlHora =
    urlHistoricoHora +
    String(timeClient.getHours()) +
    ".json";

  firebasePutInt(urlHora, lotacaoRU);
}

// =====================================================
// TEMPO ESTIMADO DE ESPERA
// =====================================================

void limparSaidas() {
  qtdSaidas = 0;

  for (int i = 0; i < MAX_SAIDAS; i++) {
    saidas[i] = 0;
  }
}

void guardarSaida(unsigned long epoch) {
  if (qtdSaidas < MAX_SAIDAS) {
    saidas[qtdSaidas] = epoch;
    qtdSaidas++;
    return;
  }

  for (int i = 0; i < MAX_SAIDAS - 1; i++) {
    saidas[i] = saidas[i + 1];
  }

  saidas[MAX_SAIDAS - 1] = epoch;
}

int calcularEsperaMinutos() {

  // Abaixo da capacidade de referência existe vaga.
  if (lotacaoRU < CAPACIDADE) {
    return 0;
  }

  // Precisamos de duas saídas para calcular um intervalo.
  if (qtdSaidas < 2) {
    return -1;
  }

  unsigned long agora = timeClient.getEpochTime();
  unsigned long ultimaSaida = saidas[qtdSaidas - 1];

  // Se a última saída é muito antiga, não mostramos
  // uma estimativa possivelmente enganosa.
  if (
    agora > ultimaSaida &&
    agora - ultimaSaida > LIMITE_INTERVALO_SAIDA
  ) {
    return -1;
  }

  unsigned long soma = 0;
  int intervalosValidos = 0;

  for (int i = 1; i < qtdSaidas; i++) {

    if (saidas[i] <= saidas[i - 1]) {
      continue;
    }

    unsigned long intervalo =
      saidas[i] - saidas[i - 1];

    if (intervalo <= LIMITE_INTERVALO_SAIDA) {
      soma += intervalo;
      intervalosValidos++;
    }
  }

  if (intervalosValidos == 0) {
    return -1;
  }

  float mediaSegundos =
    (float)soma / (float)intervalosValidos;

  int minutos =
    (int)ceil(mediaSegundos / 60.0f);

  return minutos < 1 ? 1 : minutos;
}

void atualizarEspera() {
  atualizarHora();

  int estimativa = calcularEsperaMinutos();

  if (estimativa == ultimaEsperaEnviada) {
    return;
  }

  if (firebasePutInt(urlTempoEspera, estimativa)) {
    ultimaEsperaEnviada = estimativa;

    Serial.print("[ESPERA] ");

    if (estimativa < 0) {
      Serial.println("Calculando...");
    } else if (estimativa == 0) {
      Serial.println("Sem espera");
    } else {
      Serial.print("~");
      Serial.print(estimativa);
      Serial.println(" min");
    }
  }
}

// =====================================================
// CONTROLE DO DIA
// =====================================================

void carregarDadosAtuais() {
  lotacaoRU =
    max(firebaseGetInt(urlLotacao, 0), 0);

  totalEntrada =
    max(firebaseGetInt(urlTotalEntrada, 0), 0);

  picoDia =
    max(firebaseGetInt(urlPicoDia, 0), 0);

  String primeira =
    firebaseGetString(urlPrimeiraEntrada);

  primeiraEntradaRegistrada =
    primeira.length() > 0;

  Serial.println("Dados carregados do Firebase.");
  Serial.print("Lotação: ");
  Serial.println(lotacaoRU);
  Serial.print("Entradas: ");
  Serial.println(totalEntrada);
  Serial.print("Pico: ");
  Serial.println(picoDia);
}

void resetarDia() {
  Serial.println("Virada do dia detectada.");

  lotacaoRU = 0;
  totalEntrada = 0;
  picoDia = 0;
  primeiraEntradaRegistrada = false;
  estadoCatraca = 0;

  limparSaidas();

  ultimaEsperaEnviada = -999;

  firebasePutInt(urlLotacao, 0);
  firebasePutInt(urlTotalEntrada, 0);
  firebasePutInt(urlPicoDia, 0);
  firebasePutInt(urlTempoEspera, 0);

  firebaseDelete(BASE + "/primeiraEntrada.json");
  firebaseDelete(BASE + "/historicoPorHora.json");
  firebaseDelete(BASE + "/historicoLotacao.json");

  dataLocal = obterData();

  firebasePutString(
    urlDataAtual,
    dataLocal
  );
}

void iniciarDia() {
  atualizarHora();

  dataLocal = obterData();

  String dataFirebase =
    firebaseGetString(urlDataAtual);

  // Primeira vez usando esta versão:
  // registra a data sem apagar os dados já existentes.
  if (dataFirebase == "") {
    firebasePutString(urlDataAtual, dataLocal);
    carregarDadosAtuais();
    return;
  }

  if (dataFirebase != dataLocal) {
    resetarDia();
    return;
  }

  carregarDadosAtuais();
}

void verificarViradaDia() {
  static unsigned long ultimaVerificacao = 0;

  if (millis() - ultimaVerificacao < 30000) {
    return;
  }

  ultimaVerificacao = millis();

  if (!atualizarHora()) return;

  String hoje = obterData();

  if (
    hoje != "" &&
    dataLocal != "" &&
    hoje != dataLocal
  ) {
    resetarDia();
  }
}

// =====================================================
// ENTRADA / SAÍDA
// =====================================================

void registrarEntrada() {
  lotacaoRU++;
  totalEntrada++;

  atualizarHora();

  if (!primeiraEntradaRegistrada) {
    primeiraEntradaRegistrada = true;

    firebasePutString(
      urlPrimeiraEntrada,
      obterHoraMinuto()
    );
  }

  if (lotacaoRU > picoDia) {
    picoDia = lotacaoRU;
    firebasePutInt(urlPicoDia, picoDia);
  }

  firebasePutInt(urlLotacao, lotacaoRU);
  firebasePutInt(urlTotalEntrada, totalEntrada);

  registrarHistorico();
  atualizarEspera();

  Serial.print("[+] ENTRADA | Lotação: ");
  Serial.println(lotacaoRU);
}

void registrarSaida() {
  if (lotacaoRU > 0) {
    lotacaoRU--;
  }

  atualizarHora();

  guardarSaida(
    timeClient.getEpochTime()
  );

  firebasePutInt(urlLotacao, lotacaoRU);

  registrarHistorico();
  atualizarEspera();

  Serial.print("[-] SAÍDA | Lotação: ");
  Serial.println(lotacaoRU);
}

// =====================================================
// ESTADO DA CATRACA
// =====================================================

void iniciarEstado(int estado) {
  estadoCatraca = estado;
  inicioEstadoMs = millis();
}

void verificarTimeoutEstado() {
  if (estadoCatraca == 0) return;

  if (millis() - inicioEstadoMs > 3000) {
    estadoCatraca = 0;
  }
}

// =====================================================
// SETUP
// =====================================================

void setup() {
  Serial.begin(115200);

  pinMode(SENSOR_ENTRADA, INPUT);
  pinMode(SENSOR_SAIDA, INPUT);

  WiFi.setAutoReconnect(true);
  WiFi.persistent(false);

  WiFi.begin(ssid, password);

  Serial.print("Conectando ao Wi-Fi");

  while (!wifiOK()) {
    Serial.print(".");
    delay(500);
  }

  Serial.println();
  Serial.println("Wi-Fi conectado.");

  timeClient.begin();

  for (int i = 0; i < 10; i++) {
    if (atualizarHora()) break;
    delay(500);
  }

  iniciarDia();

  // Cria um ponto inicial no gráfico.
  registrarHistorico();

  // Publica o estado inicial da espera.
  atualizarEspera();

  Serial.println("Sistema pronto.");
}

// =====================================================
// LOOP
// =====================================================

void loop() {
  timeClient.update();

  verificarViradaDia();
  verificarTimeoutEstado();

  bool bloqueouEntrada =
    digitalRead(SENSOR_ENTRADA) == LOW;

  bool bloqueouSaida =
    digitalRead(SENSOR_SAIDA) == LOW;

  // ---------------- ENTRADA ----------------
  if (
    bloqueouEntrada &&
    !bloqueouSaida &&
    estadoCatraca == 0
  ) {
    iniciarEstado(1);
  }

  else if (
    bloqueouEntrada &&
    bloqueouSaida &&
    estadoCatraca == 1
  ) {
    estadoCatraca = 2;
  }

  else if (
    !bloqueouEntrada &&
    bloqueouSaida &&
    estadoCatraca == 2
  ) {
    estadoCatraca = 3;
  }

  else if (
    !bloqueouEntrada &&
    !bloqueouSaida &&
    estadoCatraca == 3
  ) {
    estadoCatraca = 0;
    registrarEntrada();
  }

  // ---------------- SAÍDA ----------------
  else if (
    !bloqueouEntrada &&
    bloqueouSaida &&
    estadoCatraca == 0
  ) {
    iniciarEstado(-1);
  }

  else if (
    bloqueouEntrada &&
    bloqueouSaida &&
    estadoCatraca == -1
  ) {
    estadoCatraca = -2;
  }

  else if (
    bloqueouEntrada &&
    !bloqueouSaida &&
    estadoCatraca == -2
  ) {
    estadoCatraca = -3;
  }

  else if (
    !bloqueouEntrada &&
    !bloqueouSaida &&
    estadoCatraca == -3
  ) {
    estadoCatraca = 0;
    registrarSaida();
  }

  // Sequência iniciada mas abandonada.
  if (
    !bloqueouEntrada &&
    !bloqueouSaida &&
    (
      estadoCatraca == 1 ||
      estadoCatraca == -1
    )
  ) {
    estadoCatraca = 0;
  }

  delay(2);
}
